#include <stdio.h>
#define max 10
int count = 1;
int adj[max][max];
int visited[max];
int queue[max];
int front=0;
int rear=-1;
int n,e;

void insert(int ele)
{
    rear++;
    queue[rear]=ele;
    bfs(ele);
}

void del()
{
    queue[front]=0;
    front++;
}

//BFS
void bfs(int v)
{
    visited[v]=count;
    while(front<=rear)
    {
        for(int j=0;j<n; j++)
        {
            if(adj[v][j]==1 && visited[j]==0)
            {
                count++;
                visited[j]=count;
                insert(j);
            }
        }
        del();
    }
}


int main()
{
    int i,j;
    int src,dest;
    printf("Enter the number of nodes:");
    scanf("%d",&n);
    // initialise visited and adjacency matrix to 0
    for(i=0; i<n; i++)
    {
        visited[i]=0;
        for(j=0; j<n; j++)
        {
            adj[i][j]=0;
        }
    }
    printf("\nEnter the number of edges:");
    scanf("%d",&e);
    //Calculating adjacency matrix
    printf("\nEnter the Edges:");
    for(i=0; i<e; i++)
    {
        printf("\nEnter Source:");
        scanf("%d",&src);
        printf("\nEnter Destination:");
        scanf("%d",&dest);
        adj[src][dest]=1;
        adj[dest][src]=1;
    }
    printf("\nAdjacency Matrix:\n");
    for(i=0; i<n; i++)
    {
        for(j=0; j<n; j++)
        {
            printf("%d \t",adj[i][j]);
        }
        printf("\n");
    }
    insert(0);
    printf("Visiting Order:");
    for(i=0; i<n; i++)
    {
        printf("\nVertex %d is visited %d\n",i,visited[i]);
    }

}
