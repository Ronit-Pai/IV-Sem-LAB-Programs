#include <stdio.h>
#include <stdlib.h>
void insertionSort(int a[], int n)
{
    for(int i=1; i<n; i++)
    {
        int v=a[i];
        int j=i-1;
        while(j>=0 && a[j]>v)
        {
            a[j+1]=a[j];
            j--;
        }
        a[j+1]=v;
    }
    return;
}
int main()
{
    int n;
    printf("Enter size of array:");
    scanf("%d",&n);
    int a[n];
    printf("\nEnter Array:");
    for(int i=0; i<n; i++)
        scanf("%d",&a[i]);
    insertionSort(a,n);
    printf("\nSorted Array:\n");
    for(int i=0; i<n; i++)
        printf("%d\t",a[i]);
    return 0;
}
