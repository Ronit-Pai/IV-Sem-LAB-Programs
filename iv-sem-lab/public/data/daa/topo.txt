#include <stdio.h>
#include <stdlib.h>
#define max 100
int adj[max][max];
int indegree[max];
int topoOrder[max];
int queue[max];
int front=0;
int rear=-1;
int n,e,fv;
int t=-1;

void insert(int ele)
{
    rear++;
    queue[rear]=ele;
    return;
}

int del()
{
    int item = queue[front];
    front++;
    return item;
}

void topologicalSort()
{
    int i,j;
    for(i=0; i<n; i++)
    {
        if(indegree[i]==0)
        {
            insert(i);
        }
    }

    while(front<=rear)
    {
        fv=del();
        t++;
        topoOrder[t]=fv;

        for(i=0; i<n; i++)
        {
            if(adj[fv][i]==1)
            {
                indegree[i]-=1;

                if(indegree[i]==0)
                {
                    insert(i);
                }
            }
        }
    }
}

int main()
{
    int i,j;
    int src,dest;

    printf("Enter the number of nodes:");
    scanf("%d",&n);

    for(i=0; i<n; i++)
    {
        topoOrder[i]=0;
        indegree[i]=0;

        for(j=0; j<n; j++)
        {
            adj[i][j]=0;
        }
    }

    printf("\nEnter the number of edges:");
    scanf("%d",&e);

    printf("\nEnter the Edges:");
    for(i=0; i<e; i++)
    {
        printf("\nEnter Source:");
        scanf("%d",&src);

        printf("\nEnter Destination:");
        scanf("%d",&dest);

        adj[src][dest]=1;
    }

    printf("\nAdjacency Matrix:\n");
    for(i=0; i<n; i++)
    {
        for(j=0; j<n; j++)
        {
            printf("%d \t",adj[i][j]);
        }
        printf("\n");
    }

    for(i=0; i<n; i++)
    {
        for(j=0; j<n; j++)
        {
            indegree[i]+=adj[j][i];
        }
    }

    topologicalSort();

    printf("Topological Order:");
    for(i=0; i<n; i++)
    {
        printf("%d \t",topoOrder[i]);
    }

    return 0;
}
