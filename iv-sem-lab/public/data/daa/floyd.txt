#include<stdio.h>
#include<stdlib.h>
int n,i,j,a[100][100],k;
int min(int a,int b)
{
  if(a<b)
    return a;
  else
    return b;
}

void floyd()
{
  for(k=0;k<n;k++)
  {
    for(i=0;i<n;i++)
    {
      for(j=0;j<n;j++)
      {
        a[i][j]=min(a[i][j],a[i][k]+a[k][j]);
      }
    }
  }
 }
int main()
{
  printf("Enter the number of vertices:");
  scanf("%d",&n);
  printf("Enter the initial matrix:\n");
  printf("NOTE: if no path then enter value of infinity as 10000\n");
  for(i=0;i<n;i++)
  {
   for(j=0;j<n;j++)
   {
    scanf("%d",&a[i][j]);
   }
  }
  floyd();
  printf("All pair shortest path matrix:\n");
  for(i=0;i<n;i++)
  {
   for(j=0;j<n;j++)
   {
    printf("%d\t",a[i][j]);
    }
    printf("\n");
  }
  return 0;
 }
