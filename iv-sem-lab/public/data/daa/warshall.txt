#include<stdio.h>
#include<stdlib.h>
int n,i,j,a[100][100],k;
void warshall()
{
  for(k=0;k<n;k++)
  {
    for(i=0;i<n;i++)
    {
      for(j=0;j<n;j++)
      {
        a[i][j]=a[i][j] || (a[i][k] && a[k][j]);
      }
    }
  }
 }
int main()
{
  printf("Enter the number of vertices:");
  scanf("%d",&n);
  printf("Enter the adjacency matrix:\n");
  for(i=0;i<n;i++)
  {
   for(j=0;j<n;j++)
   {
    scanf("%d",&a[i][j]);
   }
  }
  warshall();

  printf("Transitive Closure:\n");
  for(i=0;i<n;i++)
  {
   for(j=0;j<n;j++)
   {
    printf("%d \t",a[i][j]);
    }
    printf("\n");
  }
  return 0;
 }
